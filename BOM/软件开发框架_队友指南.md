#### 步骤 2：CubeMX 配置（关键）

按以下顺序配置外设：

**System Core 配置**：

```
SYS:
  Debug → Serial Wire (启用 SWD,占 PA13/PA14)
  Timebase Source → SysTick

RCC:
  HSE → Crystal/Ceramic Resonator (用外部 8MHz 晶振)
  LSE → Disable (RTC 不用)

Clock Configuration:
  设置主频到 170MHz
  HSE 8MHz → PLL → 170MHz 系统时钟
```

**通信外设配置**：

```
USART2 (调试串口):
  Mode: Asynchronous
  PA2: TX
  PA3: RX
  Baud: 115200, 8N1
  ★ 启用 Global interrupt (NVIC)

I2C1 (传感器总线):
  Mode: I2C
  Speed: Fast Mode (400 kHz)
  PB6: SCL
  PB7: SDA
  ★ 不启用中断 (用阻塞模式简单)

SPI1 (LCD):
  Mode: Transmit Only Master
  Data Size: 8 Bits
  Prescaler: 4 (clock = 170M/4/8 = ~5MHz, LCD 够用)
  PA5: SCK
  PA7: MOSI
  ★ PA6 (MISO) 不启用 (LCD 单向写)

FDCAN1:
  Mode: Classic CAN
  PA11: CAN_RX
  PA12: CAN_TX
  Bit Rate: 500 kbps
  Sample Point: 87.5%
```

**GPIO 配置**：

```
输出 (Output):
  PB4: LED_GREEN (GPIO_Output, Push-Pull, Low Speed, No Pull)
  PB5: LED_RED
  PB8: BUZZER_CTRL
  PB9: BQ34_VEN
  PB0: LCD_DC
  PB1: LCD_CS
  PB2: LCD_RST
  PA4: DAC_SCLK (软件 SPI)
  PA6: DAC_DIN (软件 SPI)
  PB10: DAC_CS

输入 (Input with EXTI):
  PB3: BQ_ALERT  → Mode: External Interrupt Falling, Pull-up
  PB12: KEY1     → Mode: External Interrupt Falling, Pull-up
  PB13: KEY2     → 同上
  PB14: KEY3     → 同上
  PB15: KEY4     → 同上

NVIC:
  EXTI 3:    BQ_ALERT 中断
  EXTI 12-15: 4 个按键中断
  USART2: 调试串口
```

## 阶段 1：BSP（板级支持包）配置（Week 1，1 天）

这是**最重要的设计模式**——把所有引脚和参数集中到一个文件，硬件改动只改这里。

### 1.1 创建 BSP 配置文件

在工程的 `Core/Inc/` 目录下创建 `bsp_config.h`：

```c
/**
 * bsp_config.h
 * 板级支持包 - 集中所有硬件相关配置
 * 硬件定型后只改这个文件,其他代码不动
 */

#ifndef BSP_CONFIG_H
#define BSP_CONFIG_H

#include "main.h"
#include "stm32g4xx_hal.h"

// ====================== 硬件版本 ======================
#define BOARD_VERSION       "V2.0_Formation"
#define HARDWARE_REVISION   2

// ====================== 时钟配置 ======================
#define SYSTEM_CLOCK_HZ     170000000UL   // 170 MHz

// ====================== 调试串口 ======================
#define DEBUG_UART          huart2
#define DEBUG_BAUDRATE      115200
extern UART_HandleTypeDef huart2;

// ====================== I²C 配置 ======================
#define SENSOR_I2C          hi2c1
extern I2C_HandleTypeDef hi2c1;

// ====================== I²C 设备地址 ======================
#define I2C_ADDR_BQ76920    (0x18 << 1)   // BQ76920 7-bit 地址 0x18
#define I2C_ADDR_BQ34Z100   (0x55 << 1)   // BQ34Z100 7-bit 地址 0x55  
#define I2C_ADDR_INA226     (0x40 << 1)   // INA226 7-bit 地址 0x40 (A0=A1=GND)
#define I2C_ADDR_TMP117     (0x48 << 1)   // TMP117 7-bit 地址 0x48 (ADD0=GND)

// ====================== SPI 配置 (LCD) ======================
#define LCD_SPI             hspi1
extern SPI_HandleTypeDef hspi1;

#define LCD_CS_PORT         GPIOB
#define LCD_CS_PIN          GPIO_PIN_1

#define LCD_DC_PORT         GPIOB
#define LCD_DC_PIN          GPIO_PIN_0

#define LCD_RST_PORT        GPIOB
#define LCD_RST_PIN         GPIO_PIN_2

#define LCD_WIDTH           320       // 3.5" 屏分辨率
#define LCD_HEIGHT          480

// ====================== DAC8552 软件 SPI ======================
#define DAC_SCLK_PORT       GPIOA
#define DAC_SCLK_PIN        GPIO_PIN_4

#define DAC_DIN_PORT        GPIOA
#define DAC_DIN_PIN         GPIO_PIN_6

#define DAC_CS_PORT         GPIOB
#define DAC_CS_PIN          GPIO_PIN_10

#define DAC_VREF_VOLTS      3.3f      // REF3033 输出 3.3V
#define DAC_RESOLUTION      65536     // 16-bit

// ====================== LED ======================
#define LED_GREEN_PORT      GPIOB
#define LED_GREEN_PIN       GPIO_PIN_4

#define LED_RED_PORT        GPIOB
#define LED_RED_PIN         GPIO_PIN_5

// ====================== 蜂鸣器 ======================
#define BUZZER_PORT         GPIOB
#define BUZZER_PIN          GPIO_PIN_8

// ====================== 按键 ======================
#define KEY1_PIN            GPIO_PIN_12  // EXTI12
#define KEY2_PIN            GPIO_PIN_13  // EXTI13
#define KEY3_PIN            GPIO_PIN_14  // EXTI14
#define KEY4_PIN            GPIO_PIN_15  // EXTI15

// ====================== BQ76920 ALERT ======================
#define BQ_ALERT_PIN        GPIO_PIN_3   // EXTI3

// ====================== CAN ======================
#define CAN_HANDLE          hfdcan1
extern FDCAN_HandleTypeDef hfdcan1;
#define CAN_BAUDRATE        500000      // 500 kbps

// ====================== 4G 模组 (预留) ======================
// 当前未实现,但接口已预留
// #define MODEM_UART          huart4
// #define MODEM_PWRKEY_PORT   GPIOA
// #define MODEM_PWRKEY_PIN    GPIO_PIN_8

// ====================== 物理常数 (硬件参数) ======================
#define RSNS_OHM            0.010f      // 采样电阻 10mΩ
#define BQ34_BAT_DIV_RATIO  0.0433f     // BQ34Z100 BAT 分压比 (1.5M+68k 分压)
#define INA226_LSB_uA       (2500)      // INA226 电流分辨率 µA/LSB (按 Cal 寄存器配置)

// ====================== 系统参数 ======================
#define MAX_CELLS           5           // 5 节电池
#define CELL_NUM            5

#define BATTERY_NOMINAL_V   3.7f        // 标称电压
#define BATTERY_FULL_V      4.20f       // 满电
#define BATTERY_EMPTY_V     2.50f       // 截止
#define BATTERY_CAPACITY_AH 2.5f        // 设计容量

// 化成参数
#define FORMATION_CHARGE_C_RATE   0.5f  // 0.5C 充电
#define FORMATION_DISCHARGE_C_RATE 0.5f // 0.5C 放电
#define FORMATION_CV_END_CURRENT  0.05f // CV 阶段截止电流 (0.05C)

#endif // BSP_CONFIG_H
```

### 1.2 修改的好处

之后无论硬件怎么调整：
- 引脚改了 → 只改 `bsp_config.h` 中的 `_PIN` / `_PORT` 宏
- I²C 地址变了 → 只改 `I2C_ADDR_xxx` 宏
- 屏幕分辨率变了 → 只改 `LCD_WIDTH` / `LCD_HEIGHT`

**驱动代码完全不动**。

### 2.3 驱动模板（以 TMP117 为例）

`Core/Inc/tmp117.h`:

```c
#ifndef TMP117_H
#define TMP117_H

#include <stdint.h>
#include "bsp_config.h"

// 寄存器定义
#define TMP117_REG_TEMP_RESULT  0x00
#define TMP117_REG_CONFIG       0x01
#define TMP117_REG_DEVICE_ID    0x0F

#define TMP117_DEVICE_ID        0x0117

// 函数原型
uint8_t TMP117_Init(void);
float TMP117_ReadTemperature(void);

#endif
```

`Core/Src/tmp117.c`:

```c
#include "tmp117.h"
#include <stdio.h>

uint8_t TMP117_Init(void)
{
    uint8_t buf[2];
    HAL_StatusTypeDef status;
    
    // 读 Device ID 寄存器
    status = HAL_I2C_Mem_Read(&SENSOR_I2C, I2C_ADDR_TMP117,
                               TMP117_REG_DEVICE_ID,
                               I2C_MEMADD_SIZE_8BIT,
                               buf, 2, 100);
    
    if (status != HAL_OK) {
        printf("[TMP117] I2C error: %d\r\n", status);
        return 0;
    }
    
    uint16_t id = (buf[0] << 8) | buf[1];
    if ((id & 0x0FFF) != TMP117_DEVICE_ID) {
        printf("[TMP117] Unexpected ID: 0x%04X\r\n", id);
        return 0;
    }
    
    printf("[TMP117] Found, ID=0x%04X\r\n", id);
    return 1;
}

float TMP117_ReadTemperature(void)
{
    uint8_t buf[2];
    HAL_StatusTypeDef status;
    
    status = HAL_I2C_Mem_Read(&SENSOR_I2C, I2C_ADDR_TMP117,
                               TMP117_REG_TEMP_RESULT,
                               I2C_MEMADD_SIZE_8BIT,
                               buf, 2, 100);
    
    if (status != HAL_OK) {
        return -999.0f;
    }
    
    int16_t raw = (int16_t)((buf[0] << 8) | buf[1]);
    return raw * 0.0078125f;  // 1 LSB = 7.8125 m°C
}
```

**测试代码**（写到 main.c 里）：


### 2.4 INA226 驱动

`Core/Inc/ina226.h`:

```c
#ifndef INA226_H
#define INA226_H

#include "bsp_config.h"

// 寄存器
#define INA226_REG_CONFIG       0x00
#define INA226_REG_SHUNT_V      0x01
#define INA226_REG_BUS_V        0x02
#define INA226_REG_POWER        0x03
#define INA226_REG_CURRENT      0x04
#define INA226_REG_CALIBRATION  0x05

// 配置常量 (datasheet 公式)
// Cal = 0.00512 / (Current_LSB × Rshunt)
// 期望 Current_LSB = 2.5 mA → Cal = 0.00512 / (0.0025 × 0.010) = 204800 = 0x32000
// 但 16-bit 寄存器最大 0xFFFF,所以 Current_LSB 至少要 100 µA
// 改: Current_LSB = 100 µA = 0.0001 A → Cal = 0.00512 / (0.0001 × 0.01) = 5120 = 0x1400
#define INA226_CAL_VALUE        0x1400      // Calibration 寄存器值
#define INA226_CURRENT_LSB_UA   100         // 1 LSB = 100 µA

uint8_t INA226_Init(void);
int32_t INA226_ReadCurrent_uA(void);     // 返回当前电流 (µA, 有符号)
float INA226_ReadBusVoltage_V(void);     // 返回总线电压 (V)

#endif
```

`Core/Src/ina226.c`:

```c
#include "ina226.h"
#include <stdio.h>

static HAL_StatusTypeDef ina226_write_reg(uint8_t reg, uint16_t value)
{
    uint8_t buf[2] = { (value >> 8) & 0xFF, value & 0xFF };
    return HAL_I2C_Mem_Write(&SENSOR_I2C, I2C_ADDR_INA226,
                              reg, I2C_MEMADD_SIZE_8BIT,
                              buf, 2, 100);
}

static HAL_StatusTypeDef ina226_read_reg(uint8_t reg, uint16_t *value)
{
    uint8_t buf[2];
    HAL_StatusTypeDef s = HAL_I2C_Mem_Read(&SENSOR_I2C, I2C_ADDR_INA226,
                                            reg, I2C_MEMADD_SIZE_8BIT,
                                            buf, 2, 100);
    if (s == HAL_OK) {
        *value = (buf[0] << 8) | buf[1];
    }
    return s;
}

uint8_t INA226_Init(void)
{
    HAL_StatusTypeDef s;
    
    // 1. 配置寄存器: 16次平均 + 1.1ms 采样时间 + 连续模式
    // bits[15:12]=0001 (reset bit), [11:9]=011 (16 avg), 
    // [8:6]=100 (1.1ms shunt), [5:3]=100 (1.1ms bus), [2:0]=111 (cont)
    s = ina226_write_reg(INA226_REG_CONFIG, 0x4527);
    if (s != HAL_OK) {
        printf("[INA226] Config write failed\r\n");
        return 0;
    }
    
    // 2. 校准寄存器
    s = ina226_write_reg(INA226_REG_CALIBRATION, INA226_CAL_VALUE);
    if (s != HAL_OK) return 0;
    
    printf("[INA226] Initialized, LSB=%d uA\r\n", INA226_CURRENT_LSB_UA);
    return 1;
}

int32_t INA226_ReadCurrent_uA(void)
{
    uint16_t raw;
    if (ina226_read_reg(INA226_REG_CURRENT, &raw) != HAL_OK) {
        return 0;
    }
    
    int16_t signed_raw = (int16_t)raw;  // 转 16-bit 有符号
    return (int32_t)signed_raw * INA226_CURRENT_LSB_UA;
}

float INA226_ReadBusVoltage_V(void)
{
    uint16_t raw;
    if (ina226_read_reg(INA226_REG_BUS_V, &raw) != HAL_OK) {
        return 0.0f;
    }
    return raw * 0.00125f;  // 1 LSB = 1.25 mV
}
```

测试：

```c

while (1)
{
    int32_t i_uA = INA226_ReadCurrent_uA();
    float v = INA226_ReadBusVoltage_V();
    printf("V=%.3fV, I=%d.%03d mA\r\n", 
           v, i_uA/1000, abs(i_uA%1000));
    HAL_Delay(500);
}
```

### 2.6 DAC8552 软件 SPI 驱动

DAC8552 用软件 bit-bang SPI（不占用硬件 SPI 资源）：

`Core/Inc/dac8552.h`:

```c
#ifndef DAC8552_H
#define DAC8552_H

#include "bsp_config.h"

#define DAC_CHANNEL_A   0
#define DAC_CHANNEL_B   1

void DAC8552_Init(void);
void DAC8552_Write(uint8_t channel, uint16_t value);

// 高级函数: 直接设定输出电压 (0 ~ 3.3V)
void DAC8552_SetVoltage(uint8_t channel, float voltage);

// 高级函数: 设定恒流源目标电流 (mA)
// 内部根据 OPA 增益和 Rsns 计算 DAC 值
void DAC8552_SetCurrent(uint8_t channel, float current_mA);

#endif
```

`Core/Src/dac8552.c`:

```c
#include "dac8552.h"

// 软件 SPI 引脚操作宏
#define DAC_SCLK_HIGH() HAL_GPIO_WritePin(DAC_SCLK_PORT, DAC_SCLK_PIN, GPIO_PIN_SET)
#define DAC_SCLK_LOW()  HAL_GPIO_WritePin(DAC_SCLK_PORT, DAC_SCLK_PIN, GPIO_PIN_RESET)
#define DAC_DIN_HIGH()  HAL_GPIO_WritePin(DAC_DIN_PORT, DAC_DIN_PIN, GPIO_PIN_SET)
#define DAC_DIN_LOW()   HAL_GPIO_WritePin(DAC_DIN_PORT, DAC_DIN_PIN, GPIO_PIN_RESET)
#define DAC_CS_HIGH()   HAL_GPIO_WritePin(DAC_CS_PORT, DAC_CS_PIN, GPIO_PIN_SET)
#define DAC_CS_LOW()    HAL_GPIO_WritePin(DAC_CS_PORT, DAC_CS_PIN, GPIO_PIN_RESET)

static void dac8552_send_byte(uint8_t byte)
{
    for (int i = 7; i >= 0; i--) {
        DAC_SCLK_LOW();
        if (byte & (1 << i)) DAC_DIN_HIGH();
        else                  DAC_DIN_LOW();
        DAC_SCLK_HIGH();
    }
}

void DAC8552_Init(void)
{
    DAC_CS_HIGH();
    DAC_SCLK_LOW();
    DAC_DIN_LOW();
    
    // 上电默认输出 0V
    DAC8552_Write(DAC_CHANNEL_A, 0);
    DAC8552_Write(DAC_CHANNEL_B, 0);
}

void DAC8552_Write(uint8_t channel, uint16_t value)
{
    // 命令字节: 写并加载 DAC 通道
    uint8_t cmd = (channel == DAC_CHANNEL_A) ? 0x10 : 0x24;
    
    DAC_CS_LOW();
    dac8552_send_byte(cmd);
    dac8552_send_byte((value >> 8) & 0xFF);
    dac8552_send_byte(value & 0xFF);
    DAC_CS_HIGH();
}

void DAC8552_SetVoltage(uint8_t channel, float voltage)
{
    if (voltage < 0) voltage = 0;
    if (voltage > DAC_VREF_VOLTS) voltage = DAC_VREF_VOLTS;
    
    uint16_t value = (uint16_t)(voltage / DAC_VREF_VOLTS * (DAC_RESOLUTION - 1));
    DAC8552_Write(channel, value);
}

// 计算: 设定电流 → DAC 输出电压
// 假设 OPA 增益 11x, Rsns=10mΩ
// 实际电流 I → Rsns 上电压 = I × 0.01V
// OPA 输出 = I × 0.01 × 11 = I × 0.11V (假设增益)
// DAC 输出应等于 I × 0.11V
// (具体增益值由 OPA 反馈电阻决定,Week 5 标定)
void DAC8552_SetCurrent(uint8_t channel, float current_mA)
{
    float voltage = current_mA * 0.001f * RSNS_OHM * 11.0f;  // 11x 增益
    DAC8552_SetVoltage(channel, voltage);
}
```

```c
    STATE_IDLE,
    STATE_PRE_CHARGE,
    STATE_CC_CHARGE,
```

`Core/Src/formation.c` 关键函数：

```c
void Formation_Tick(formation_context_t *ctx)
{
    // 1. 读取当前状态
    ctx->pack_voltage_V = read_pack_voltage();
    ctx->current_mA = INA226_ReadCurrent_uA() / 1000.0f;
    ctx->temperature_C = TMP117_ReadTemperature();
    
    uint32_t elapsed = HAL_GetTick() - ctx->state_start_tick;
    
    // 2. 状态转移
    switch (ctx->state) {
    
    case STATE_CC_CHARGE:
        // 设定充电电流
        DAC8552_SetCurrent(DAC_CHANNEL_B, ctx->charge_current_mA);
        
        // 累计电量 (mA·s = mC,除以 3.6 = mAh)
        ctx->charged_mAh += ctx->current_mA * 0.001f / 3.6f;  // 假设 1ms 调用一次
        
        // 转入 CV
        if (ctx->pack_voltage_V >= ctx->charge_max_voltage_V) {
            transition_to(ctx, STATE_CV_CHARGE);
        }
        break;
    
    case STATE_CV_CHARGE:
        // 软件维持电压 = 4.2V (恒压模式,DAC 反馈控制)
        // 这里需要 PID 闭环控制
        cv_charge_pid_control(ctx);
        ctx->charged_mAh += ctx->current_mA * 0.001f / 3.6f;
        
        // 截止条件: 电流降到 0.05C
        if (ctx->current_mA <= ctx->charge_min_current_mA) {
            DAC8552_SetCurrent(DAC_CHANNEL_B, 0);  // 关闭充电
            transition_to(ctx, STATE_REST_AFTER_CHARGE);
        }
        break;
    
    case STATE_REST_AFTER_CHARGE:
        DAC8552_SetCurrent(DAC_CHANNEL_A, 0);
        DAC8552_SetCurrent(DAC_CHANNEL_B, 0);
        if (elapsed >= ctx->rest_duration_ms) {
            transition_to(ctx, STATE_CC_DISCHARGE);
        }
        break;
    
    case STATE_CC_DISCHARGE:
        DAC8552_SetCurrent(DAC_CHANNEL_A, ctx->discharge_current_mA);
        ctx->discharged_mAh += -ctx->current_mA * 0.001f / 3.6f; // 放电电流取反
        
        if (ctx->pack_voltage_V <= ctx->discharge_min_voltage_V) {
            DAC8552_SetCurrent(DAC_CHANNEL_A, 0);
            transition_to(ctx, STATE_REST_AFTER_DISCHARGE);
        }
        break;
    
    case STATE_REST_AFTER_DISCHARGE:
        if (elapsed >= ctx->rest_duration_ms) {
            transition_to(ctx, STATE_COMPLETE);
            printf("[Formation] DONE: charged=%.1f mAh, discharged=%.1f mAh\r\n",
                   ctx->charged_mAh, ctx->discharged_mAh);
        }
        break;
    
    case STATE_COMPLETE:
    case STATE_IDLE:
    case STATE_ERROR:
        // 无主动动作,等待用户命令
        break;
    }
}
```

### 3.3 CV 模式 PID 闭环

CV (恒压) 模式比 CC (恒流) 复杂，需要 PID 控制：

```c
// 简单 PID 控制器
static float cv_pid_kp = 0.5f;
static float cv_pid_ki = 0.01f;
static float cv_pid_integral = 0;

static void cv_charge_pid_control(formation_context_t *ctx)
{
    // 目标: 维持 pack_voltage = 4.2V
    float error = ctx->charge_max_voltage_V - ctx->pack_voltage_V;
    
    cv_pid_integral += error * cv_pid_ki;
    
    // 防止 integral windup
    if (cv_pid_integral > ctx->charge_current_mA) 
        cv_pid_integral = ctx->charge_current_mA;
    if (cv_pid_integral < 0) 
        cv_pid_integral = 0;
    
    float current_setpoint = error * cv_pid_kp + cv_pid_integral;
    
    // 限幅
    if (current_setpoint < 0) current_setpoint = 0;
    if (current_setpoint > ctx->charge_current_mA) 
        current_setpoint = ctx->charge_current_mA;
    
    DAC8552_SetCurrent(DAC_CHANNEL_B, current_setpoint);
}
```

PID 参数 `kp/ki` 需要 Week 6 实测调试。

### 5.1 CAN 报文格式（自定义）

```
ID: 0x100  (BMS 状态广播帧)
DLC: 8
Data:
  [0-1]: Pack Voltage (mV, big-endian)
  [2-3]: Current (mA, signed, big-endian)
  [4]:   State (0-7, 当前化成阶段)

  [6-7]: Charged Capacity (mAh, big-endian)
```

```c
void can_send_status(formation_context_t *ctx)
{
    FDCAN_TxHeaderTypeDef header = {
        .Identifier = 0x100,
        .IdType = FDCAN_STANDARD_ID,
        .TxFrameType = FDCAN_DATA_FRAME,
        .DataLength = FDCAN_DLC_BYTES_8,
        .ErrorStateIndicator = FDCAN_ESI_ACTIVE,
        .BitRateSwitch = FDCAN_BRS_OFF,
        .FDFormat = FDCAN_CLASSIC_CAN,
        .TxEventFifoControl = FDCAN_NO_TX_EVENTS,
        .MessageMarker = 0
    };
    
    uint16_t v = (uint16_t)(ctx->pack_voltage_V * 1000);
    int16_t i = (int16_t)ctx->current_mA;
    uint16_t cap = (uint16_t)ctx->charged_mAh;
    
    uint8_t data[8] = {
        (v >> 8) & 0xFF, v & 0xFF,
        (i >> 8) & 0xFF, i & 0xFF,
        (uint8_t)ctx->state,
        (int8_t)ctx->temperature_C,
        (cap >> 8) & 0xFF, cap & 0xFF,
    };
    
    HAL_FDCAN_AddMessageToTxFifoQ(&CAN_HANDLE, &header, data);
}
```
